FRONTEND INTEGRATION CODE
=========================

JavaScript/TypeScript Integration:

```javascript
class DmensionAgent {
  constructor(baseUrl) {
    this.baseUrl = baseUrl;
    this.apiKey = null; // Not required for public access
  }

  // Get all available shapes
  async getShapes() {
    const response = await fetch(`${this.baseUrl}/shapes/registry`);
    return response.json();
  }

  // Generate 3D geometry for a shape
  async generateGeometry(shapeId, parameters = {}) {
    const response = await fetch(`${this.baseUrl}/compute/parametric`, {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({shapeId, parameters, quality: 'high'})
    });
    return response.json();
  }

  // AI-powered shape analysis
  async analyzeShape(shapeId, analysisType = 'optimization') {
    const response = await fetch(`${this.baseUrl}/ai-agent/shapes/${shapeId}/analyze`, {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({analysisType, targetAudience: 'general', depth: 0.8})
    });
    return response.json();
  }

  // Export 3D model
  async export3D(shapeId, parameters, format = 'glb') {
    const response = await fetch(`${this.baseUrl}/export/universal`, {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({shapeId, parameters, format})
    });
    return response.json();
  }

  // Discover viral triggers
  async discoverTriggers(viralThreshold = 0.8) {
    const response = await fetch(`${this.baseUrl}/ai-agent/triggers/discover?viral_threshold=${viralThreshold}`);
    return response.json();
  }

  // Generate AI content
  async generateContent(shapeId, contentType = 'documentation') {
    const response = await fetch(`${this.baseUrl}/ai-agent/generate-content`, {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({shapeId, contentType, specifications: {}})
    });
    return response.json();
  }
}

// Usage Example
const agent = new DmensionAgent('https://your-repl-url.replit.dev/api');

// Get all shapes

// Generate a sphere with custom parameters
const geometry = await agent.generateGeometry('sphere', {a: 2, b: 2, c: 2});

// AI analysis
const analysis = await agent.analyzeShape('heart_chakra', 'therapeutic');

// Export as GLB
const exportResult = await agent.export3D('torus', {a: 2, b: 0.5}, 'glb');
```

INTEGRATION EXAMPLES
===================

React Component Integration:
```javascript
import { useEffect, useState } from 'react';

function MathVisualizer() {
  const [agent] = useState(() => new DmensionAgent('/api'));
  const [shapes, setShapes] = useState([]);
  const [geometry, setGeometry] = useState(null);

  useEffect(() => {
    agent.getShapes().then(setShapes);
  }, []);

  const generateShape = async (shapeId, params) => {
    const result = await agent.generateGeometry(shapeId, params);
    setGeometry(result);
  };

  return (
    <div>
      {/* Shape selector and parameter controls */}
      <ShapeControls onGenerate={generateShape} />
      <ThreeJSCanvas geometry={geometry} />
    </div>
  );
}
```
