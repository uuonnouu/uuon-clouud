  <style>  
    body, html {  
      margin: 0;  
      padding: 0;  
      background: #0e0e1a;  
      overflow: hidden;  
      font-family: 'Segoe UI', sans-serif;  
    }  
    canvas {  
      display: block;  
    }  
    .tooltip {  
      position: absolute;  
      color: white;  
      background: rgba(0, 0, 0, 0.7);  
      padding: 5px 10px;  
      border-radius: 5px;  
      font-size: 14px;  
      pointer-events: none;  
      display: none;  
    }

<body>  
<canvas id="ptCanvas"></canvas>  
<div class="tooltip" id="tooltip"></div>  
<script>  
const canvas = document.getElementById("ptCanvas");  
const ctx = canvas.getContext("2d");  
const tooltip = document.getElementById("tooltip");  
  
let width = canvas.width = window.innerWidth;  
let height = canvas.height = window.innerHeight;  
let centerX = width / 2;  
let centerY = height / 2;  
let radius = Math.min(width, height) * 0.35;  
  
const elements = [];  
const selectedElements = [];  
  
for (let Z = 1; Z <= 120; Z++) {  
  elements.push({  
    atomicNumber: Z,  
    symbol: `E${Z}`,  
    x: 0,  
    y: 0,  
    angle: 0  
  });  
}  
  
function getAngleForElement(Z) {  
  if (Z % 15 === 0) return (Z / 15) * 15;  
  if (Z % 24 === 0) return (Z / 24) * 24;  
  if (Z % 30 === 0) return (Z / 30) * 30;  
  if (Z % 40 === 0) return (Z / 40) * 40;  
  return Z * 3;  
}  
  
function layoutElements() {  
  elements.forEach(el => {  
    let angleDeg = getAngleForElement(el.atomicNumber);  
    let angleRad = angleDeg * Math.PI / 180;  
    el.angle = angleRad;  
    el.x = centerX + radius * Math.cos(angleRad);  
    el.y = centerY + radius * Math.sin(angleRad);  
  });  
}  
  
function drawScene() {  
  ctx.clearRect(0, 0, width, height);  
  for (let a = 0; a < 360; a += 15) {  
    let rad = a * Math.PI / 180;  
    ctx.beginPath();  
    ctx.moveTo(centerX, centerY);  
    ctx.lineTo(centerX + radius * Math.cos(rad), centerY + radius * Math.sin(rad));  
    ctx.strokeStyle = "rgba(100,100,255,0.1)";  
    ctx.stroke();  
  }  
  
  elements.forEach(el => {  
    ctx.beginPath();  
    ctx.arc(el.x, el.y, 14, 0, 2 * Math.PI);  
    ctx.fillStyle = selectedElements.includes(el) ? "#ff0066" : "#00aaff";  
    ctx.fill();  
    ctx.fillStyle = "#ffffff";  
    ctx.font = "12px sans-serif";  
    ctx.textAlign = "center";  
    ctx.fillText(el.symbol, el.x, el.y + 4);  
  });  
  
  if (selectedElements.length >= 2) {  
    ctx.beginPath();  
    ctx.moveTo(selectedElements[0].x, selectedElements[0].y);  
    selectedElements.slice(1).forEach(el => ctx.lineTo(el.x, el.y));  
    ctx.strokeStyle = "#ffffff88";  
    ctx.lineWidth = 2;  
    ctx.stroke();  
  }  
}  
  
canvas.addEventListener("click", (e) => {  
  const rect = canvas.getBoundingClientRect();  
  const mx = e.clientX - rect.left;  
  const my = e.clientY - rect.top;  
  
  elements.forEach(el => {  
    let dx = el.x - mx;  
    let dy = el.y - my;  
    if (dx * dx + dy * dy < 196) {  
      if (!selectedElements.includes(el)) {  
        selectedElements.push(el);  
        if (selectedElements.length > 3) selectedElements.shift();  
      }  
    }  
  });  
  drawScene();  
  updateTooltip(mx, my);  
});  
  
canvas.addEventListener("mousemove", (e) => {  
  const rect = canvas.getBoundingClientRect();  
  const mx = e.clientX - rect.left;  
  const my = e.clientY - rect.top;  
  
  let found = false;  
  elements.forEach(el => {  
    let dx = el.x - mx;  
    let dy = el.y - my;  
    if (dx * dx + dy * dy < 196) {  
      tooltip.style.left = (mx + 10) + "px";  
      tooltip.style.top = (my + 10) + "px";  
      tooltip.innerHTML = `Element ${el.symbol} (Z=${el.atomicNumber})`;  
      tooltip.style.display = "block";  
      found = true;  
    }  
  });  
  if (!found) tooltip.style.display = "none";  
});  
  
function updateTooltip(x, y) {  
  if (selectedElements.length >= 2) {  
    let compound = selectedElements.map(e => e.symbol).join("-");  
    tooltip.style.left = (x + 10) + "px";  
    tooltip.style.top = (y + 10) + "px";  
    tooltip.innerHTML = `Compound: ${compound}<br>ΔE ≈ -${100 * selectedElements.length} kJ/mol`;  
    tooltip.style.display = "block";  
  }  
}  
  
layoutElements();  
drawScene();  
  
window.addEventListener("resize", () => {  
  width = canvas.width = window.innerWidth;  
  height = canvas.height = window.innerHeight;  
  centerX = width / 2;  
  centerY = height / 2;  
  radius = Math.min(width, height) * 0.35;  
  layoutElements();  
  drawScene();  
});  
</script>  
</body>
