## TASK 18B — FIX SHAPE VISIBILITY (Default Camera + Per-Shape Defaults)

### Part 1: Smart camera positioning
In `MathVisualizer.tsx`, find `CameraAutoAdjuster` component.
It currently sets camera to (50, 25, 60) for all shapes.

Add shape-category-aware camera positioning:
```typescript
function CameraAutoAdjuster({ orbitRef, shapeType }) {
  const { camera } = useThree();
  const hasInitialized = useRef(false);
  const prevShape = useRef('');

  useEffect(() => {
    if (!orbitRef.current) return;
    // Reset camera when shape changes
    if (prevShape.current === shapeType && hasInitialized.current) return;
    prevShape.current = shapeType;
    hasInitialized.current = true;

    // Shape-category camera distances
    const distances: Record<string, [number,number,number]> = {
      // Tiny shapes (quantum, atomic scale)
      'hydrogen': [3, 1.5, 4],
      'orbital': [3, 1.5, 4],
      'qubit': [3, 1.5, 4],
      // Normal shapes
      'sphere': [8, 4, 10],
      'torus': [8, 4, 10],
      'klein': [10, 5, 12],
      // Large shapes (cosmic)
      'black_hole': [20, 10, 25],
      'schwarzschild': [20, 10, 25],
      'gravitational': [15, 8, 20],
      // Attractors (need wide view)
      'lorenz': [60, 30, 80],
      'rossler': [40, 20, 50],
      // 4D projections
      'tesseract': [10, 5, 12],
      'hopf': [8, 4, 10],
    };

    // Find matching camera position
    let pos: [number,number,number] = [50, 25, 60]; // default
    for (const [key, camPos] of Object.entries(distances)) {
      if (shapeType.includes(key)) {
        pos = camPos;
        break;
      }
    }

    camera.position.set(...pos);
    orbitRef.current.target.set(0, 0, 0);
    orbitRef.current.update();
  }, [camera, orbitRef, shapeType]);

  return null;
}
```
