# ΔMENSION — TASK #17: PARAMETER MANIFEST SYSTEM
# The fix for parameters not matching shapes

---

## THE PROBLEM

The A-Z parameter system is universal but most shapes only use A, B, C.
D through Z sliders exist but do nothing for 90% of shapes.
Users have no idea what "A" means for a Klein bottle vs a Lorenz attractor.
Parameter ranges are wrong — sphere radius doesn't need to go to 180.

## THE SOLUTION: Parameter Manifests

Add a `parameter_manifest` JSONB column to `formula_implementations`.
For each shape, this declares which parameters are active, what they mean,
and what their valid range is for that specific shape.

The UI then reads this manifest and:
- Shows only active sliders
- Labels them correctly ("Major Radius" not "A")
- Uses shape-specific min/max ranges
- Hides unused parameters instead of showing broken sliders

## GOLDEN RULES
- Do NOT touch ParametricSurface.tsx rendering logic
- Do NOT change the A-Z parameter names in SurfaceParameters type
- Do NOT change how parameters are passed to formulas
- Only add data and update the UI display layer

// Helper: get manifest for a shape, with fallback to generic
export function getParameterManifest(shapeType: string): ShapeParameterManifest | null {
  return PARAMETER_MANIFESTS[shapeType] || null;
}

// Helper: get active parameter keys for a shape
export function getActiveParameters(shapeType: string): string[] {
  const manifest = PARAMETER_MANIFESTS[shapeType];
  if (!manifest) return ['a', 'b', 'c']; // Default: show first 3
  return Object.keys(manifest.active_parameters);
}

// Helper: get label for a parameter in context of a shape
export function getParameterLabel(shapeType: string, param: string): string {
  const manifest = PARAMETER_MANIFESTS[shapeType];
  if (!manifest) return param.toUpperCase();
  const spec = manifest.active_parameters[param as keyof typeof manifest.active_parameters];
  return spec?.label || param.toUpperCase();
}

// Helper: get range for a parameter in context of a shape  
export function getParameterRange(shapeType: string, param: string): {min:number, max:number, step:number} {
  const manifest = PARAMETER_MANIFESTS[shapeType];
  if (!manifest) return { min: -26, max: 26, step: 0.01 };
  const spec = manifest.active_parameters[param as keyof typeof manifest.active_parameters];
  return spec ? { min: spec.min, max: spec.max, step: spec.step } : { min: -26, max: 26, step: 0.01 };
}

## STEP 5: Seed parameter_manifest into database

After creating `parameterManifests.ts`, run a script to populate the column:

```bash
cat > scripts/seed-manifests.ts << 'EOF'
import { db } from '../server/db';
import { formula_implementations } from '../shared/schema';
import { eq } from 'drizzle-orm';
import { PARAMETER_MANIFESTS } from '../client/src/lib/parameterManifests';

async function seedManifests() {
  let updated = 0;
  for (const [shape_type, manifest] of Object.entries(PARAMETER_MANIFESTS)) {
    try {
      await db.update(formula_implementations)
        .set({ parameter_manifest: manifest.active_parameters })
        .where(eq(formula_implementations.shape_type, shape_type));
      updated++;
    } catch(e) {
      console.error(`Failed for ${shape_type}:`, e);
    }
  }
  console.log(`Updated ${updated} shape manifests`);
}
seedManifests().catch(console.error);
EOF
npx tsx scripts/seed-manifests.ts
```
