### Discovery 1: Opacity bug source identified
Lines 2184, 2193, 2227 in ParametricSurface.tsx have transparent:true at opacity 1.0.
This puts fully-opaque meshes in the transparent render queue = z-fighting + depth errors.

### Discovery 2: GPU animation already exists and is being ignored
animationShaders.ts has a complete vertex shader with twist/inflate/wave/morph uniforms.
These run ON THE GPU — zero geometry rebuilds, zero CPU cost, true 60fps animation.
AutonomousAnimation.tsx changes React state every frame = geometry rebuild every frame = slow.
The fix: use animationShaders.ts uniforms instead of React state for animation.

### Discovery 3: The spectral shader is commercially unique
surfaceShader.ts contains CIE 1931 physics-accurate spectral color + Fresnel + 
Phi-harmonic golden ratio balance — all in GLSL. This combination is rare.
It belongs in a shader marketplace. It belongs as an API endpoint.

### Step 2: Create GpuAnimation component
Create `client/src/components/GpuAnimation.tsx`:

```tsx
import { useRef, useEffect } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';
import { createAnimationUniforms, AnimationUniforms } from '../lib/animationShaders';

interface GpuAnimationProps {
  meshRef: React.RefObject<THREE.Mesh>;
  enabled: boolean;
  mode: 'breathe' | 'wave' | 'twist' | 'morph';
  intensity: number;
}

export default function GpuAnimation({ meshRef, enabled, mode, intensity }: GpuAnimationProps) {
  const uniformsRef = useRef<AnimationUniforms>(createAnimationUniforms());
  
  useFrame((state) => {
    if (!enabled || !meshRef.current) return;
    
    const time = state.clock.elapsedTime;
    const uniforms = uniformsRef.current;
    
    // Update time uniform — this drives all animation
    uniforms.uTime.value = time;
    uniforms.uAmplitude.value = intensity;
    
    if (mode === 'breathe') {
      uniforms.uScale.value = 1.0 + Math.sin(time * 0.8) * 0.1 * intensity;
      uniforms.uInflation.value = 1.0 + Math.sin(time * 0.6) * 0.05 * intensity;
      uniforms.uTwist.value = 0;
      uniforms.uWaveStrength.value = 0.1 * intensity;
    } else if (mode === 'wave') {
      uniforms.uScale.value = 1.0;
      uniforms.uWaveStrength.value = intensity;
      uniforms.uFrequency.value = 2.0 + Math.sin(time * 0.3) * 1.0;
      uniforms.uInflation.value = 1.0;
    } else if (mode === 'twist') {
      uniforms.uTwist.value = Math.sin(time * 0.5) * intensity * 2.0;
      uniforms.uScale.value = 1.0;
      uniforms.uWaveStrength.value = 0.2 * intensity;
    } else if (mode === 'morph') {
      uniforms.uMorphProgress.value = (Math.sin(time * 0.4) + 1) / 2;
      uniforms.uTwist.value = Math.sin(time * 0.3) * intensity;
      uniforms.uScale.value = 1.0 + Math.sin(time * 0.7) * 0.15 * intensity;
      uniforms.uWaveStrength.value = intensity * 0.5;
      uniforms.uInflation.value = 1.0 + Math.sin(time * 0.9) * 0.1 * intensity;
    }
    
    // Apply uniforms to mesh material if it's a ShaderMaterial
    if (meshRef.current.material && 
        (meshRef.current.material as THREE.ShaderMaterial).uniforms) {
      const mat = meshRef.current.material as THREE.ShaderMaterial;
      Object.assign(mat.uniforms, uniformsRef.current);
      mat.needsUpdate = false; // uniforms don't need needsUpdate
    }
  });
  
  return null;
}
```

/**
 * POST /api/shapes/:shapeType/uvmap
 * Returns UV coordinate arrays for texture mapping
 */
router.get('/uvmap/:shapeType', async (req: Request, res: Response) => {
  const { shapeType } = req.params;
  const segments = parseInt(req.query.segments as string) || 32;
  
  try {
    // Get shape formula from DB
    const shapes = await db.select({
      equation_x: formula_implementations.equation_x_formula,
      equation_y: formula_implementations.equation_y_formula,
      equation_z: formula_implementations.equation_z_formula,
      uv_domain: formula_implementations.uv_domain,
    }).from(formula_implementations)
      .where(eq(formula_implementations.shape_type, shapeType))
      .limit(1);
    
    if (!shapes.length) {
      return res.status(404).json({ error: `Shape '${shapeType}' not found` });
    }
    
    const shape = shapes[0];
    const domain = shape.uv_domain as any;
    
    // Generate UV grid
    const uMin = domain?.uMin ?? 0;
    const uMax = domain?.uMax ?? Math.PI * 2;
    const vMin = domain?.vMin ?? 0;
    const vMax = domain?.vMax ?? Math.PI;
    
    const uvCoords: number[][] = [];
    for (let i = 0; i <= segments; i++) {
      for (let j = 0; j <= segments; j++) {
        const u = uMin + (uMax - uMin) * (i / segments);
        const v = vMin + (vMax - vMin) * (j / segments);
        uvCoords.push([i / segments, j / segments]); // Normalized 0-1
      }
    }
    
    res.json({
      shape_type: shapeType,
      uv_domain: { uMin, uMax, vMin, vMax },
      segments,
      total_vertices: uvCoords.length,
      uv_coordinates: uvCoords,
      format: 'Array of [u, v] pairs normalized 0-1',
      usage: 'Apply as TEXCOORD_0 to any mesh with the same vertex count'
    });
    
  } catch (error) {
    res.status(500).json({ error: 'UV generation failed', details: String(error) });
  }
});

/**
 * GET /api/color/spectrum?nm=550
 * Convert wavelength to RGB using CIE 1931
 */
router.get('/spectrum', (req: Request, res: Response) => {
  const nm = parseFloat(req.query.nm as string);
  
  if (isNaN(nm) || nm < 380 || nm > 740) {
    return res.status(400).json({ 
      error: 'Wavelength must be 380-740nm (visible spectrum)',
      example: '/api/color/spectrum?nm=550'
    });
  }
  
  // CIE 1931 2° Standard Observer (same as surfaceShader.ts)
  let r = 0, g = 0, b = 0;
  
  if (nm >= 380 && nm < 440) {
    r = -(nm - 440) / (440 - 380); g = 0; b = 1;
  } else if (nm >= 440 && nm < 490) {
    r = 0; g = (nm - 440) / (490 - 440); b = 1;
  } else if (nm >= 490 && nm < 510) {
    r = 0; g = 1; b = -(nm - 510) / (510 - 490);
  } else if (nm >= 510 && nm < 580) {
    r = (nm - 510) / (580 - 510); g = 1; b = 0;
  } else if (nm >= 580 && nm < 645) {
    r = 1; g = -(nm - 645) / (645 - 580); b = 0;
  } else if (nm >= 645 && nm <= 740) {
    r = 1; g = 0; b = 0;
  }
  
  // Edge falloff
  let factor = 1.0;
  if (nm >= 380 && nm < 420) factor = 0.3 + 0.7 * (nm - 380) / (420 - 380);
  else if (nm > 700 && nm <= 740) factor = 0.3 + 0.7 * (740 - nm) / (740 - 700);
  
  r *= factor; g *= factor; b *= factor;
  
  const toHex = (v: number) => Math.round(v * 255).toString(16).padStart(2, '0');
  
  res.json({
    wavelength_nm: nm,
    rgb: { r: parseFloat(r.toFixed(4)), g: parseFloat(g.toFixed(4)), b: parseFloat(b.toFixed(4)) },
    rgb_255: { r: Math.round(r * 255), g: Math.round(g * 255), b: Math.round(b * 255) },
    hex: `#${toHex(r)}${toHex(g)}${toHex(b)}`,
    standard: 'CIE 1931 2° Standard Observer',
    color_region: nm < 450 ? 'violet/blue' : nm < 495 ? 'blue/cyan' : nm < 570 ? 'green' : nm < 620 ? 'yellow/orange' : 'red',
  });
});

/**
 * POST /api/color/tonemap
 * HDR to display-ready RGB with Reinhard or ACES
 */
router.post('/tonemap', (req: Request, res: Response) => {
  const { r, g, b, exposure = 1.0, method = 'reinhard' } = req.body;
  
  if (r === undefined || g === undefined || b === undefined) {
    return res.status(400).json({ error: 'Provide r, g, b values (can exceed 1.0 for HDR)' });
  }
  
  let dr = r * exposure, dg = g * exposure, db = b * exposure;
  
  if (method === 'reinhard') {
    dr = dr / (1 + dr); dg = dg / (1 + dg); db = db / (1 + db);
  } else if (method === 'aces') {
    const a = 2.51, bv = 0.03, c = 2.43, d = 0.59, e = 0.14;
    dr = Math.max(0, Math.min(1, (dr * (a * dr + bv)) / (dr * (c * dr + d) + e)));
    dg = Math.max(0, Math.min(1, (dg * (a * dg + bv)) / (dg * (c * dg + d) + e)));
    db = Math.max(0, Math.min(1, (db * (a * db + bv)) / (db * (c * db + d) + e)));
  }
  
  res.json({
    input: { r, g, b, exposure, method },
    output: { r: parseFloat(dr.toFixed(4)), g: parseFloat(dg.toFixed(4)), b: parseFloat(db.toFixed(4)) },
    hex: `#${Math.round(dr*255).toString(16).padStart(2,'0')}${Math.round(dg*255).toString(16).padStart(2,'0')}${Math.round(db*255).toString(16).padStart(2,'0')}`,
  });
});
