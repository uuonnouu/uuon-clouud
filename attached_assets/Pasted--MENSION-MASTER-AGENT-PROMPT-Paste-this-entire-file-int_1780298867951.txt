```bash
import { db } from '../server/db';
import { formula_implementations, shape_embeddings } from '../shared/schema';
import { eq } from 'drizzle-orm';

async function seedEmbeddings() {
  const shapes = await db.select({
    shape_type: formula_implementations.shape_type,
    category: formula_implementations.category,
    complexity_score: formula_implementations.complexity_score,
    equation_x: formula_implementations.equation_x_formula,
    equation_y: formula_implementations.equation_y_formula,
    equation_z: formula_implementations.equation_z_formula,
    genus: formula_implementations.genus,
    orientable: formula_implementations.orientable,
    mathematical_foundation: formula_implementations.mathematical_foundation,
  }).from(formula_implementations)
    .where(eq(formula_implementations.is_verified, true));

  console.log(`Processing ${shapes.length} verified shapes...`);

  for (const shape of shapes) {
    // Create mathematical feature vector from known properties
    const features = {
      complexity: shape.complexity_score / 10,
      has_trig: (shape.equation_x?.includes('sin') || shape.equation_x?.includes('cos')) ? 1 : 0,
      has_exp: shape.equation_x?.includes('exp') ? 1 : 0,
      is_orientable: shape.orientable ? 1 : 0,
      genus_normalized: (shape.genus || 0) / 10,
    };

    const categoryMap: Record<string, number> = {
      'topology': 1, 'fractals': 2, 'quantum-mechanics': 3, 'general-relativity': 4,
      'cryptography': 5, '4d-hyperdimensional': 6, 'molecular-biology': 7,
      'sacred-geometry': 8, 'theory-of-everything': 9, 'algorithms': 10,
    };
    const cat_id = (categoryMap[shape.category] || 0) / 10;

    // Simple embedding vector — 16 dimensions
    const embedding = [
      features.complexity, features.has_trig, features.has_exp,
      features.is_orientable, features.genus_normalized, cat_id,
      shape.complexity_score / 10, Math.random() * 0.01, // small noise
      features.complexity * features.has_trig,
      features.is_orientable * cat_id,
      features.genus_normalized * features.complexity,
      cat_id * features.has_trig,
      features.complexity * cat_id,
      features.has_exp * features.complexity,
      features.genus_normalized * cat_id,
      features.is_orientable * features.complexity,
    ];

    await db.insert(shape_embeddings).values({
      shape_type: shape.shape_type,
      embedding_vector: embedding,
      mathematical_features: features,
      symmetry_signature: `cat_${shape.category}_g${shape.genus || 0}`,
      topology_signature: `orientable_${shape.orientable}_genus_${shape.genus || 0}`,
      equation_complexity: shape.complexity_score,
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
    }).onConflictDoUpdate({
      target: shape_embeddings.shape_type,
      set: { updated_at: new Date().toISOString() }
    });
  }

  console.log('Done seeding shape_embeddings');
}
```

```bash
async function seedTopology() {
  const shapes = await db.select().from(formula_implementations);
  console.log(`Seeding topology for ${shapes.length} shapes...`);
  let filled = 0, defaulted = 0;

  for (const shape of shapes) {
    const known = KNOWN_TOPOLOGY[shape.shape_type];
    const b0 = known?.b0 ?? 1;
    const b1 = known?.b1 ?? 0;
    const b2 = known?.b2 ?? (shape.orientable ? 1 : 0);
    const euler = known?.euler ?? (b0 - b1 + b2);
    const genus = known?.genus ?? shape.genus ?? 0;
    const phase = known?.phase ?? 'surface';

    await db.insert(gip_identity_metrics).values({
      shape_type: shape.shape_type,
      entropy_value: Math.log2(Math.max(1, shape.complexity_score)),
      identity_preservation: 1.0,
      consciousness_level: 0,
      geometric_phase: phase,
      dimensional_complexity: 3,
      betti_0: b0,
      betti_1: b1,
      betti_2: b2,
      euler_characteristic: euler,
      hausdorff_dimension: shape.orientable ? 2.0 : 2.0,
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
    }).onConflictDoUpdate({
      target: gip_identity_metrics.shape_type,
      set: { updated_at: new Date().toISOString() }
    });

    known ? filled++ : defaulted++;
  }
  console.log(`Done: ${filled} with known values, ${defaulted} with defaults`);
}
```

### 2C. Fill shape_relationships using token similarity
```bash
cat > scripts/seed-relationships.ts << 'EOF'
import { db } from '../server/db';
import { formula_implementations, shape_relationships } from '../shared/schema';

async function seedRelationships() {
  const shapes = await db.select({
    shape_type: formula_implementations.shape_type,
    category: formula_implementations.category,
    genus: formula_implementations.genus,
    orientable: formula_implementations.orientable,
    complexity_score: formula_implementations.complexity_score,
    mathematical_foundation: formula_implementations.mathematical_foundation,
  }).from(formula_implementations);

  console.log(`Building relationships for ${shapes.length} shapes...`);
  let count = 0;

  for (let i = 0; i < shapes.length; i++) {
    for (let j = i + 1; j < shapes.length; j++) {
      const a = shapes[i], b = shapes[j];
      // Same category = structural relationship
      if (a.category === b.category) {
        const similarity = 1 - Math.abs((a.complexity_score - b.complexity_score)) / 10;
        if (similarity > 0.5) {
          try {
            await db.insert(shape_relationships).values({
              shape_a: a.shape_type,
              shape_b: b.shape_type,
              relationship_type: 'same_category',
              similarity_score: similarity,
              shared_mathematics: { category: a.category },
              connection_explanation: `Both are ${a.category} shapes with similar complexity`,
              discovered_by: 'automated_seeding',
              discovery_method: 'category_matching',
              validation_status: 'pending',
              created_at: new Date().toISOString(),
            }).onConflictDoNothing();
            count++;
          } catch(e) {}
        }
      }
      // Same genus and orientability = topological relationship
      if (a.genus === b.genus && a.orientable === b.orientable && a.genus !== null) {
        try {
          await db.insert(shape_relationships).values({
            shape_a: a.shape_type,
            shape_b: b.shape_type,
            relationship_type: 'topological_equivalence',
            similarity_score: 0.85,
            shared_mathematics: { genus: a.genus, orientable: a.orientable },
            connection_explanation: `Both genus-${a.genus} ${a.orientable ? 'orientable' : 'non-orientable'} surfaces`,
            discovered_by: 'automated_seeding',
            discovery_method: 'topology_matching',
            validation_status: 'verified',
            created_at: new Date().toISOString(),
          }).onConflictDoNothing();
          count++;
        } catch(e) {}
      }
    }
  }
  console.log(`Created ${count} shape relationships`);
}

seedRelationships().catch(console.error);
EOF
```
